// Assets/Scripts/FPSCharacterController.cs
using UnityEngine;
using Nexo.Feature.Platform.Models;

public class FPSCharacterController : MonoBehaviour, INexoBusinessLogic
{
    [Header("Movement Settings")]
    public float moveSpeed = 5f;
    public float jumpHeight = 3f;
    public float mouseSensitivity = 2f;
    
    // Cross-domain business logic patterns
    public MovementPattern MovementPattern { get; set; }
    public InputPattern InputPattern { get; set; }
    public PhysicsPattern PhysicsPattern { get; set; }
    
    private CharacterController controller;
    private Vector3 velocity;
    private bool isGrounded;
    
    void Start()
    {
        controller = GetComponent<CharacterController>();
        
        // Initialize cross-domain patterns
        MovementPattern = new MovementPattern
        {
            MoveSpeed = moveSpeed,
            JumpHeight = jumpHeight,
            IsGrounded = isGrounded
        };
        
        InputPattern = new InputPattern
        {
            MouseSensitivity = mouseSensitivity,
            SupportedInputs = new[] { "WASD", "Mouse", "Gamepad" }
        };
        
        PhysicsPattern = new PhysicsPattern
        {
            Gravity = Physics.gravity.y,
            GroundCheckDistance = 0.1f
        };
    }
    
    void Update()
    {
        HandleMovement();
        HandleMouseLook();
        HandleJump();
    }
    
    private void HandleMovement()
    {
        // Business logic that can be reused across platforms
        float x = Input.GetAxis("Horizontal");
        float z = Input.GetAxis("Vertical");
        
        Vector3 move = transform.right * x + transform.forward * z;
        controller.Move(move * MovementPattern.MoveSpeed * Time.deltaTime);
    }
    
    private void HandleMouseLook()
    {
        // Business logic that can be reused across platforms
        float mouseX = Input.GetAxis("Mouse X") * InputPattern.MouseSensitivity;
        float mouseY = Input.GetAxis("Mouse Y") * InputPattern.MouseSensitivity;
        
        transform.Rotate(Vector3.up * mouseX);
        Camera.main.transform.Rotate(Vector3.left * mouseY);
    }
    
    private void HandleJump()
    {
        // Business logic that can be reused across platforms
        if (Input.GetButtonDown("Jump") && isGrounded)
        {
            velocity.y = Mathf.Sqrt(MovementPattern.JumpHeight * -2f * PhysicsPattern.Gravity);
        }
        
        velocity.y += PhysicsPattern.Gravity * Time.deltaTime;
        controller.Move(velocity * Time.deltaTime);
    }
}

// web/src/controllers/CanvasCharacterController.ts
import { MovementPattern, InputPattern, PhysicsPattern } from '@nexo/platform-models';

export class CanvasCharacterController implements INexoBusinessLogic {
    private canvas: HTMLCanvasElement;
    private ctx: CanvasRenderingContext2D;
    private character: { x: number; y: number; width: number; height: number };
    
    // Same cross-domain business logic patterns as Unity
    public MovementPattern: MovementPattern;
    public InputPattern: InputPattern;
    public PhysicsPattern: PhysicsPattern;
    
    constructor(canvas: HTMLCanvasElement) {
        this.canvas = canvas;
        this.ctx = canvas.getContext('2d')!;
        
        // Initialize with same patterns as Unity
        this.MovementPattern = new MovementPattern({
            moveSpeed: 5,      // Same as Unity
            jumpHeight: 3,     // Same as Unity
            isGrounded: true   // Same as Unity
        });
        
        this.InputPattern = new InputPattern({
            mouseSensitivity: 2,  // Same as Unity
            supportedInputs: ['WASD', 'Mouse', 'Touch'] // Adapted for web
        });
        
        this.PhysicsPattern = new PhysicsPattern({
            gravity: -9.81,   // Same physics as Unity
            groundCheckDistance: 0.1  // Same as Unity
        });
        
        this.character = { x: 100, y: 100, width: 32, height: 32 };
        this.setupEventListeners();
    }
    
    private setupEventListeners() {
        // Web-specific input handling using same business logic
        document.addEventListener('keydown', (e) => this.handleInput(e));
        this.canvas.addEventListener('mousemove', (e) => this.handleMouseLook(e));
    }
    
    private handleMovement() {
        // Same business logic as Unity, adapted for canvas
        const keys = this.getPressedKeys();
        let moveX = 0, moveY = 0;
        
        if (keys.includes('KeyW')) moveY -= this.MovementPattern.moveSpeed;
        if (keys.includes('KeyS')) moveY += this.MovementPattern.moveSpeed;
        if (keys.includes('KeyA')) moveX -= this.MovementPattern.moveSpeed;
        if (keys.includes('KeyD')) moveX += this.MovementPattern.moveSpeed;
        
        this.character.x += moveX;
        this.character.y += moveY;
    }
    
    private handleMouseLook(event: MouseEvent) {
        // Same business logic as Unity, adapted for canvas
        const rect = this.canvas.getBoundingClientRect();
        const mouseX = (event.clientX - rect.left) * this.InputPattern.mouseSensitivity;
        const mouseY = (event.clientY - rect.top) * this.InputPattern.mouseSensitivity;
        
        // Apply same rotation logic as Unity
        this.rotateCharacter(mouseX, mouseY);
    }
    
    private handleJump() {
        // Same business logic as Unity, adapted for canvas
        if (this.isGrounded() && this.getPressedKeys().includes('Space')) {
            this.character.y -= this.MovementPattern.jumpHeight;
        }
    }
    
    public update() {
        // Same update loop as Unity
        this.handleMovement();
        this.handleJump();
        this.draw();
    }
    
    private draw() {
        // Web-specific rendering
        this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
        this.ctx.fillRect(this.character.x, this.character.y, this.character.width, this.character.height);
    }
}

// mobile/src/controllers/TouchCharacterController.ts
import { MovementPattern, InputPattern, PhysicsPattern } from '@nexo/platform-models';

export class TouchCharacterController implements INexoBusinessLogic {
    private character: { x: number; y: number; width: number; height: number };
    private touchControls: HTMLDivElement;
    
    // Same cross-domain business logic patterns as Unity and Web
    public MovementPattern: MovementPattern;
    public InputPattern: InputPattern;
    public PhysicsPattern: PhysicsPattern;
    
    constructor() {
        // Initialize with same patterns as Unity and Web
        this.MovementPattern = new MovementPattern({
            moveSpeed: 5,      // Same as Unity and Web
            jumpHeight: 3,     // Same as Unity and Web
            isGrounded: true   // Same as Unity and Web
        });
        
        this.InputPattern = new InputPattern({
            mouseSensitivity: 2,  // Same as Unity and Web
            supportedInputs: ['Touch', 'Gyroscope', 'Accelerometer'] // Adapted for mobile
        });
        
        this.PhysicsPattern = new PhysicsPattern({
            gravity: -9.81,   // Same physics as Unity and Web
            groundCheckDistance: 0.1  // Same as Unity and Web
        });
        
        this.character = { x: 100, y: 100, width: 32, height: 32 };
        this.setupTouchControls();
    }
    
    private setupTouchControls() {
        // Mobile-specific input handling using same business logic
        this.touchControls = document.getElementById('touch-controls') as HTMLDivElement;
        
        // Virtual joystick for movement
        const joystick = document.getElementById('joystick') as HTMLDivElement;
        joystick.addEventListener('touchstart', (e) => this.handleTouchStart(e));
        joystick.addEventListener('touchmove', (e) => this.handleTouchMove(e));
        
        // Jump button
        const jumpButton = document.getElementById('jump-button') as HTMLButtonElement;
        jumpButton.addEventListener('touchstart', () => this.handleJump());
    }
    
    private handleMovement(touchX: number, touchY: number) {
        // Same business logic as Unity and Web, adapted for touch
        const centerX = this.touchControls.offsetWidth / 2;
        const centerY = this.touchControls.offsetHeight / 2;
        
        const deltaX = (touchX - centerX) / centerX;
        const deltaY = (touchY - centerY) / centerY;
        
        this.character.x += deltaX * this.MovementPattern.moveSpeed;
        this.character.y += deltaY * this.MovementPattern.moveSpeed;
    }
    
    private handleJump() {
        // Same business logic as Unity and Web, adapted for touch
        if (this.isGrounded()) {
            this.character.y -= this.MovementPattern.jumpHeight;
        }
    }
    
    private handleTouchStart(event: TouchEvent) {
        // Mobile-specific touch handling
        event.preventDefault();
        const touch = event.touches[0];
        this.handleMovement(touch.clientX, touch.clientY);
    }
    
    private handleTouchMove(event: TouchEvent) {
        // Mobile-specific touch handling
        event.preventDefault();
        const touch = event.touches[0];
        this.handleMovement(touch.clientX, touch.clientY);
    }
    
    public update() {
        // Same update loop as Unity and Web
        this.handleJump();
        this.draw();
    }
    
    private draw() {
        // Mobile-specific rendering
        const canvas = document.getElementById('game-canvas') as HTMLCanvasElement;
        const ctx = canvas.getContext('2d')!;
        
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        ctx.fillRect(this.character.x, this.character.y, this.character.width, this.character.height);
    }
}
